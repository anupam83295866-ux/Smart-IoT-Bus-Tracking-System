/**
 * MU Bus Tracker - ESP32 + SIM7600 Built-in GPS Sketch (Bus 02)
 * Target: ESP32 DevKit V1 / NodeMCU-32S
 * Modules: SIM7600 4G LTE Module (Using Internal GNSS/GPS)
 * 
 * Description:
 * 1. Uses SIM7600 internal GNSS/GPS receiver (AT+CGPS=1 & AT+CGPSINFO) for location tracking.
 * 2. Uploads telemetry to Middleman Server using ESP32 HardwareSerial (UART2) for BUS_ID "bus_02".
 * 3. No external Neo-6M GPS module required for Bus 02!
 * 4. Requires middleman.js / start_tracker.bat running on PC!
 * 
 * ----------------------------------------------------
 * WIRING INSTRUCTIONS FOR ESP32 & SIM7600:
 * 
 * SIM7600 4G Module:
 *  - VCC        ->  5V External Power Source (Requires 5V at 2A peak current)
 *  - GND        ->  ESP32 GND (Common Ground is REQUIRED)
 *  - TXD        ->  ESP32 GPIO 16 (GSM RX2)
 *  - RXD        ->  ESP32 GPIO 17 (GSM TX2)
 *  - GPS Antenna -> Connect Passive/Active GPS Antenna to SIM7600 GPS IPEX/SMA port
 * ----------------------------------------------------
 */

// --- configurations ---
const String APN = "gpinternet"; // Cellular APN (e.g. gpinternet, blweb, banglalink)

// Replace with your localtunnel / tunnelmole URL generated by start_tracker.bat
String MIDDLEMAN_HOST = "https://rctjur-ip-27-147-226-102.tunnelmole.net";
const String BUS_ID = "bus_02"; // BUS 02 configuration
const int SEND_INTERVAL_MS = 3000; // Telemetry transfer frequency (3 seconds)

// --- pin mapping for ESP32 ---
const int GSM_RX_PIN = 16;  // ESP32 GPIO 16 connected to SIM7600 TXD
const int GSM_TX_PIN = 17;  // ESP32 GPIO 17 connected to SIM7600 RXD

// --- ESP32 HardwareSerial port ---
HardwareSerial gsmSerial(2); // UART2 for SIM7600 4G

unsigned long lastSendTime = 0;

// --- function prototypes ---
void initializeSIM7600Network();
void syncGSMBaudRate();
bool getSIM7600GPSLocation(double &lat, double &lng, float &speed);
void uploadTelemetryToMiddleman(double lat, double lng, float speed);
String sendATCommand(String command, const int timeout);

void setup() {
  Serial.begin(115200);
  gsmSerial.begin(115200, SERIAL_8N1, GSM_RX_PIN, GSM_TX_PIN);
  
  Serial.println("==========================================");
  Serial.println("MU Bus Tracker - ESP32 SIM7600 4G (Bus 02) Booting");
  Serial.println("==========================================");
  
  delay(5000); // Allow SIM7600 to complete hardware power-up
  
  // Auto-baud rate detection and configuration for SIM7600 (default 115200)
  syncGSMBaudRate();
  
  sendATCommand("AT", 1000);
  sendATCommand("ATE0", 1000); // Echo off
  sendATCommand("AT+CPSI?", 1000); // Query network status
  
  // Turn ON SIM7600 Internal GPS/GNSS engine
  Serial.println("Turning ON SIM7600 Internal GNSS/GPS Engine...");
  sendATCommand("AT+CGPS=1", 2000);
  
  // Initialize SIM7600 4G LTE Network Context
  initializeSIM7600Network();
}

void loop() {
  if (millis() - lastSendTime > SEND_INTERVAL_MS) {
    double lat = 0.0;
    double lng = 0.0;
    float speed = 0.0;
    
    // Try reading location from SIM7600 internal GPS
    bool hasFix = getSIM7600GPSLocation(lat, lng, speed);
    
    if (hasFix) {
      Serial.print("SUCCESS! Real SIM7600 4G GPS Fix Obtained! Lat: ");
      Serial.print(lat, 6);
      Serial.print(", Lng: ");
      Serial.print(lng, 6);
      Serial.print(", Speed: ");
      Serial.print(speed, 1);
      Serial.println(" km/h");
      
      uploadTelemetryToMiddleman(lat, lng, speed);
    } else {
      Serial.println(">> [Indoor Mode Activated] Uploading Test Location for Bus 02...");
      
      // Fallback Indoor Test Coordinates for Bus 02 (Sylhet)
      double indoorTestLat = 24.894213;
      double indoorTestLng = 91.865540;
      float indoorTestSpeed = 0.0;
      
      uploadTelemetryToMiddleman(indoorTestLat, indoorTestLng, indoorTestSpeed);
    }
    
    lastSendTime = millis();
  }
}

void syncGSMBaudRate() {
  long baudRates[] = {115200, 9600, 38400, 19200};
  Serial.println("Syncing SIM7600 Baud Rate...");
  
  for (int i = 0; i < 4; i++) {
    long currentBaud = baudRates[i];
    gsmSerial.begin(currentBaud, SERIAL_8N1, GSM_RX_PIN, GSM_TX_PIN);
    delay(100);
    
    while(gsmSerial.available()) gsmSerial.read(); // Clear buffer
    
    gsmSerial.println("AT");
    delay(400);
    
    String response = "";
    while (gsmSerial.available()) {
      response += (char)gsmSerial.read();
    }
    
    if (response.indexOf("OK") >= 0) {
      Serial.print("SIM7600 found at ");
      Serial.print(currentBaud);
      Serial.println(" baud!");
      
      if (currentBaud != 115200) {
        Serial.println("Reconfiguring SIM7600 to lock at 115200 baud...");
        gsmSerial.println("AT+IPR=115200");
        delay(200);
        gsmSerial.begin(115200, SERIAL_8N1, GSM_RX_PIN, GSM_TX_PIN);
        delay(100);
      }
      return;
    }
  }
  
  Serial.println("Could not auto-detect SIM7600 baud rate. Falling back to 115200.");
  gsmSerial.begin(115200, SERIAL_8N1, GSM_RX_PIN, GSM_TX_PIN);
}

void initializeSIM7600Network() {
  Serial.println("Initializing 4G LTE Data Context on SIM7600...");
  
  sendATCommand("AT+CGREG?", 2000); // Check network registration
  sendATCommand("AT+CGDCONT=1,\"IP\",\"" + APN + "\"", 2000); // APN setup
  
  String netResp = sendATCommand("AT+NETOPEN", 3000);
  if (netResp.indexOf("NETOPEN: 0") >= 0 || netResp.indexOf("OK") >= 0 || netResp.indexOf("opened") >= 0) {
    Serial.println("SIM7600 Network Opened Successfully!");
  } else {
    Serial.println("Network Open command executed.");
  }
  
  sendATCommand("AT+IPADDR", 2000);
}

// Helper function to query and parse SIM7600 built-in GPS (+CGPSINFO)
bool getSIM7600GPSLocation(double &lat, double &lng, float &speed) {
  String response = sendATCommand("AT+CGPSINFO", 1000);
  
  int idx = response.indexOf("+CGPSINFO:");
  if (idx < 0) return false;
  
  String data = response.substring(idx + 10);
  data.trim();
  
  // Check if GPS info is valid (not empty/commas only e.g. ",,,,,,,,")
  if (data.startsWith(",") || data.length() < 10) {
    return false;
  }
  
  // Parse CSV: lat,N/S,lon,E/W,date,time,alt,speed,course
  // Example: 2453.6528,N,09151.9324,E,010826,170500.0,40.5,0.0,0.0
  int comma1 = data.indexOf(',');
  int comma2 = data.indexOf(',', comma1 + 1);
  int comma3 = data.indexOf(',', comma2 + 1);
  int comma4 = data.indexOf(',', comma3 + 1);
  int comma5 = data.indexOf(',', comma4 + 1);
  int comma6 = data.indexOf(',', comma5 + 1);
  int comma7 = data.indexOf(',', comma6 + 1);
  int comma8 = data.indexOf(',', comma7 + 1);
  
  if (comma1 < 0 || comma2 < 0 || comma3 < 0 || comma4 < 0) return false;
  
  String rawLat = data.substring(0, comma1);
  String ns     = data.substring(comma1 + 1, comma2);
  String rawLng = data.substring(comma2 + 1, comma3);
  String ew     = data.substring(comma3 + 1, comma4);
  String rawSpd = (comma7 >= 0 && comma8 >= 0) ? data.substring(comma7 + 1, comma8) : "0.0";
  
  if (rawLat.length() < 4 || rawLng.length() < 5) return false;
  
  // Convert DDMM.MMMM to Decimal Degrees
  double latDeg = rawLat.substring(0, 2).toDouble();
  double latMin = rawLat.substring(2).toDouble();
  lat = latDeg + (latMin / 60.0);
  if (ns.equalsIgnoreCase("S")) lat = -lat;
  
  double lngDeg = rawLng.substring(0, 3).toDouble();
  double lngMin = rawLng.substring(3).toDouble();
  lng = lngDeg + (lngMin / 60.0);
  if (ew.equalsIgnoreCase("W")) lng = -lng;
  
  // Convert knots to km/h
  speed = rawSpd.toFloat() * 1.852;
  
  return true;
}

void uploadTelemetryToMiddleman(double lat, double lng, float speed) {
  // Ensure network context is active
  String netStatus = sendATCommand("AT+NETOPEN?", 1500);
  if (netStatus.indexOf("+NETOPEN: 1") < 0 && netStatus.indexOf("NETOPEN: 1") < 0) {
    Serial.println("Network disconnected. Re-opening SIM7600 Network...");
    initializeSIM7600Network();
    delay(1000);
  }
  
  Serial.println("Uploading telemetry to Middleman Server for " + BUS_ID + "...");
  
  sendATCommand("AT+HTTPTERM", 500);
  delay(100);
  sendATCommand("AT+HTTPINIT", 1000);
  
  String url = MIDDLEMAN_HOST + "/middleman.php?i=" + BUS_ID + 
               "&a=" + String(lat, 6) + "&o=" + String(lng, 6) + "&s=" + String((int)speed);
               
  sendATCommand("AT+HTTPPARA=\"URL\",\"" + url + "\"", 1000);
  
  Serial.println("Executing HTTP GET for " + BUS_ID + "...");
  sendATCommand("AT+HTTPACTION=0", 6000);
  
  sendATCommand("AT+HTTPREAD=0,500", 1500);
  sendATCommand("AT+HTTPTERM", 1000);
  Serial.println("SUCCESS! Telemetry uploaded for " + BUS_ID + ".");
}

// --- AT command helper ---
String sendATCommand(String command, const int timeout) {
  String response = "";
  gsmSerial.println(command);
  
  long int time = millis();
  while ((time + timeout) > millis()) {
    while (gsmSerial.available()) {
      char c = gsmSerial.read();
      response += c;
    }
  }
  
  Serial.print("AT Sent: ");
  Serial.println(command);
  Serial.print("SIM7600 Resp: ");
  Serial.println(response);
  
  return response;
}

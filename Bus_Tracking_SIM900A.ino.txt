/**
 * MU Bus Tracker - ESP32 + SIM900A + Neo-6M GPS Sketch (Bus 01)
 * Target: ESP32 DevKit V1 / NodeMCU-32S
 * Modules: Neo-6M GPS Module + SIM900A GSM Module
 * 
 * Description:
 * 1. Reads GPS coordinates from Neo-6M GPS module using ESP32 HardwareSerial (UART1).
 * 2. Uploads telemetry to Middleman Server using ESP32 HardwareSerial (UART2).
 * 3. Requires middleman.js / start_tracker.bat running on PC!
 * 
 * ----------------------------------------------------
 * WIRING INSTRUCTIONS FOR ESP32:
 * 
 * 1. Neo-6M GPS Module:
 *    - VCC  ->  3.3V or 5V
 *    - GND  ->  ESP32 GND
 *    - TXD  ->  ESP32 GPIO 27 (GPS RX1)
 *    - RXD  ->  ESP32 GPIO 26 (GPS TX1)
 * 
 * 2. SIM900A GSM Module:
 *    - VCC  ->  5V External Power Source (5V at 2A peak current)
 *    - GND  ->  ESP32 GND (Common Ground is REQUIRED)
 *    - TXD  ->  ESP32 GPIO 16 (GSM RX2)
 *    - RXD  ->  ESP32 GPIO 17 (GSM TX2 - Use 1k/2k resistor voltage divider)
 * ----------------------------------------------------
 */

#include <TinyGPS++.h>

// --- configurations ---
const String APN = "gpinternet"; // Cellular APN (e.g. gpinternet, blweb)

// Replace with your localtunnel URL generated by start_tracker.bat
String MIDDLEMAN_HOST = "6gifr2-ip-27-147-226-102.tunnelmole.net";
const String BUS_ID = "bus_01";
const int SEND_INTERVAL_MS = 5000; // Telemetry transfer frequency (5 seconds for 2G SIM900A)

// --- pin mapping for ESP32 ---
const int GPS_RX_PIN = 27;  // ESP32 GPIO 27 connected to GPS TXD
const int GPS_TX_PIN = 26;  // ESP32 GPIO 26 connected to GPS RXD
const int GSM_RX_PIN = 16;  // ESP32 GPIO 16 connected to SIM900A TXD
const int GSM_TX_PIN = 17;  // ESP32 GPIO 17 connected to SIM900A RXD

// --- ESP32 HardwareSerial ports (No SoftwareSerial needed on ESP32!) ---
HardwareSerial gpsSerial(1); // UART1 for GPS
HardwareSerial gsmSerial(2); // UART2 for SIM900A
TinyGPSPlus gps;

unsigned long lastSendTime = 0;

// --- function prototypes ---
void initializeGPRS();
void syncGSMBaudRate();
void uploadTelemetryToMiddleman(double lat, double lng, float speed);
String sendATCommand(String command, const int timeout);

void setup() {
  Serial.begin(115200);
  gpsSerial.begin(9600, SERIAL_8N1, GPS_RX_PIN, GPS_TX_PIN);
  gsmSerial.begin(9600, SERIAL_8N1, GSM_RX_PIN, GSM_TX_PIN);
  
  Serial.println("==========================================");
  Serial.println("MU Bus Tracker - ESP32 SIM900A Middleman Booting");
  Serial.println("==========================================");
  
  delay(5000); // Allow SIM900A to complete hardware power-up
  
  // Auto-baud rate detection and configuration lock for SIM900A at 9600
  syncGSMBaudRate();
  
  sendATCommand("AT", 1000);
  sendATCommand("ATE0", 1000); // Echo off
  sendATCommand("AT+CFUN=1", 1000); // Full functionality
  
  // Set GPRS parameters
  initializeGPRS();
}

void loop() {
  // Read GPS coordinates for 2 seconds to parse fresh satellite NMEA sentences
  unsigned long gpsStart = millis();
  int bytesRead = 0;
  while (millis() - gpsStart < 2000) {
    while (gpsSerial.available() > 0) {
      char c = gpsSerial.read();
      gps.encode(c);
      bytesRead++;
    }
  }

  // Telemetry loop
  if (millis() - lastSendTime > SEND_INTERVAL_MS) {
    Serial.print("[GPS DIAGNOSTIC] Bytes: ");
    Serial.print(bytesRead);
    Serial.print(" | Processed: ");
    Serial.print(gps.charsProcessed());
    Serial.print(" | Valid: ");
    Serial.print(gps.location.isValid() ? "YES" : "NO");
    Serial.print(" | Age: ");
    Serial.print(gps.location.age());
    Serial.println(" ms");

    // Allow age up to 10 seconds (10000ms)
    if (gps.location.isValid() && gps.location.age() < 10000) {
      double latitude = gps.location.lat();
      double longitude = gps.location.lng();
      float speedKmph = gps.speed.kmph();
      
      Serial.print("SUCCESS! Real GPS Satellite Fix Obtained! Lat: ");
      Serial.print(latitude, 6);
      Serial.print(", Lng: ");
      Serial.print(longitude, 6);
      Serial.print(", Speed: ");
      Serial.print(speedKmph, 1);
      Serial.println(" km/h");
      
      uploadTelemetryToMiddleman(latitude, longitude, speedKmph);
    } else {
      Serial.println(">> [Indoor Mode Activated] Uploading Test Location to Middleman (bus_01)...");
      
      // Fallback Indoor Test Coordinates (Zindabazar, Sylhet)
      double indoorTestLat = 24.894213;
      double indoorTestLng = 91.865540;
      float indoorTestSpeed = 0.0;
      
      uploadTelemetryToMiddleman(indoorTestLat, indoorTestLng, indoorTestSpeed);
    }
    lastSendTime = millis();
  }
}

void syncGSMBaudRate() {
  long baudRates[] = {9600, 115200, 19200, 38400};
  Serial.println("Syncing GSM Baud Rate...");
  
  for (int i = 0; i < 4; i++) {
    long currentBaud = baudRates[i];
    gsmSerial.begin(currentBaud, SERIAL_8N1, GSM_RX_PIN, GSM_TX_PIN);
    delay(100);
    
    // Clear buffer
    while(gsmSerial.available()) gsmSerial.read();
    
    gsmSerial.println("AT");
    delay(400);
    
    String response = "";
    while (gsmSerial.available()) {
      response += (char)gsmSerial.read();
    }
    
    if (response.indexOf("OK") >= 0) {
      Serial.print("GSM found at ");
      Serial.print(currentBaud);
      Serial.println(" baud!");
      
      if (currentBaud != 9600) {
        Serial.println("Reconfiguring GSM to lock at 9600 baud...");
        gsmSerial.println("AT+IPR=9600");
        delay(200);
        gsmSerial.begin(9600, SERIAL_8N1, GSM_RX_PIN, GSM_TX_PIN);
        delay(100);
        gsmSerial.println("AT&W"); // Save profile
        delay(200);
      }
      return;
    }
  }
  
  Serial.println("Could not auto-detect GSM baud rate. Falling back to 9600.");
  gsmSerial.begin(9600, SERIAL_8N1, GSM_RX_PIN, GSM_TX_PIN);
}

void initializeGPRS() {
  Serial.println("Initializing GPRS Context on SIM900A...");
  
  sendATCommand("AT+CGATT=1", 2000); // Attach GPRS service
  sendATCommand("AT+CIPSHUT", 1500); // Close current IP session if any
  
  sendATCommand("AT+SAPBR=3,1,\"CONTYPE\",\"GPRS\"", 1500);
  sendATCommand("AT+SAPBR=3,1,\"APN\",\"" + APN + "\"", 1500);
  sendATCommand("AT+SAPBR=1,1", 3000); // Open GPRS context
  sendATCommand("AT+SAPBR=2,1", 1500); // Query local GPRS IP address
}

void uploadTelemetryToMiddleman(double lat, double lng, float speed) {
  // Verify GPRS connection is still active
  String statusResponse = sendATCommand("AT+SAPBR=2,1", 1500);
  if (statusResponse.indexOf("0.0.0.0") >= 0 || statusResponse.indexOf("ERROR") >= 0) {
    Serial.println("GPRS disconnected or module rebooted. Reconnecting...");
    initializeGPRS();
    delay(1000);
  }
  
  Serial.println("Uploading telemetry to Middleman Server (bus_01)...");
  
  // 1. Terminate stale HTTP context
  sendATCommand("AT+HTTPTERM", 500);
  delay(100);

  // 2. Initialize HTTP Client
  sendATCommand("AT+HTTPINIT", 1000);
  sendATCommand("AT+HTTPPARA=\"CID\",1", 1000);
  
  String url = "http://" + MIDDLEMAN_HOST + "/middleman.php?i=" + BUS_ID + 
               "&a=" + String(lat, 6) + "&o=" + String(lng, 6) + "&s=" + String((int)speed);
               
  sendATCommand("AT+HTTPPARA=\"URL\",\"" + url + "\"", 1000);
  
  // Execute HTTP GET (Action = 0)
  Serial.println("Executing HTTP GET to Middleman Server...");
  sendATCommand("AT+HTTPACTION=0", 6000);
  
  // Read Response & Terminate HTTP Context
  sendATCommand("AT+HTTPREAD", 1500);
  sendATCommand("AT+HTTPTERM", 1000);
  Serial.println("SUCCESS! Telemetry sent to Middleman Server (HTTP 200 OK).");
}

// --- AT command helper ---
String sendATCommand(String command, const int timeout) {
  String response = "";
  gsmSerial.println(command);
  
  long int time = millis();
  while ((time + timeout) > millis()) {
    while (gsmSerial.available()) {
      char c = gsmSerial.read();
      response += c;
    }
  }
  
  Serial.print("AT Sent: ");
  Serial.println(command);
  Serial.print("GSM Resp: ");
  Serial.println(response);
  
  return response;
}
